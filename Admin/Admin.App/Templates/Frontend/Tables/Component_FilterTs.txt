
import { Component, OnInit } from '@angular/core';
import { NgbActiveModal } from '@ng-bootstrap/ng-bootstrap';
import { FormsModule } from '@angular/forms';
import { CommonModule } from '@angular/common';
import { IQueryResponse, ILookup } from '@bfs/_shared/interfaces';
//Template_Start_Component_AutoComplete_1
import { IAutoComplete, AutoCompleteHelper } from '@bfs/_shared/helpers/auto-complete.class';
//Template_End_Component_AutoComplete_1
import { I[ComponentNameCapital]Filter } from './[ComponentFileName].shared';

@Component({
    selector: 'app-[ComponentFileName]-filter',
    imports: [FormsModule, CommonModule],
    templateUrl: './[ComponentFileName].filter.component.html'
    //styles: ``
})
export class [ComponentNameCapital]FilterComponent implements OnInit {

    public result = {} as I[ComponentNameCapital]Filter;
    
    // Define look ups
    //Template_Start_Field_Lookup_1
    public [LookupNameCapital]Options:  any[] = [];
    //Template_End_Field_Lookup_1

    // Define autocomplete fields
    //Template_Start_Field_AutoComplete_1
    [AutoCompleteSmall]Auto: AutoCompleteHelper;
    //Template_End_Field_AutoComplete_1

    // Define range filters
    //Template_Start_FilterField_Range_1
    public [FilterRangeName]From: [FrontendFieldType] | undefined;
    public [FilterRangeName]To: [FrontendFieldType] | undefined;
    //Template_End_FilterField_Range_1

    public isLoading: any = { list: false, view: false, save: false, lookups: false, autoComplete: false };
    public submit: boolean = false;
    public errorMessage: string = '';
    public infoMessage: string = '';
    public currentOperation: string = '';
    public parent: any;
    //---------------------------------------------------------
    constructor(public activeModal: NgbActiveModal) {
    //Template_Start_Field_AutoComplete_2
        this.[AutoCompleteSmall]Auto = new AutoCompleteHelper({ queryUrl: "/[AutoCompleteCapital]/list", fieldName: '[AutoCompleteCapital]', control: null, id: '', name: '', showDropDown: false, options: [], isLoading: false, isInitial: true } as IAutoComplete);
    //Template_End_Field_AutoComplete_2
    }

    async ngOnInit(): Promise<void> {
        this.result = this.parent.queryRequest.filter || {};
        await this.getLookups();
        await this.setAutoComplete();
        // Initialize range filters if not set
        //Template_Start_FilterField_Range_2
        this.[FilterRangeName]From = this.result.[FilterRangeName]?.from;
        this.[FilterRangeName]To   = this.result.[FilterRangeName]?.to;
        //Template_End_FilterField_Range_2
    }
    //---------------------------------------------------------
    async getLookups(): Promise<void> {
        let target = '';
        //Template_Start_Field_Lookup_2
        target = "/[LookupNameCapital]/list";
        (await this.parent.apiService.post(target,  {pageSize:50})).subscribe({
            next: (response: IQueryResponse) => {
                this.[LookupNameCapital]Options = response.items;
                this.isLoading.list = false;
            },
                error: (err: any) => {
                this.errorMessage = err.message || 'An error occurred while fetching [DisplayName] data.';
                this.isLoading.list = false;
            }
        });
        //Template_End_Field_Lookup_2
    }
    //---------------------------------------------------------
    async setAutoComplete() {
    //Template_Start_Field_AutoComplete_3
        this.[AutoCompleteSmall]Auto.setupFilter(this.parent.apiService, this.result);
    //Template_End_Field_AutoComplete_3
}
//---------------------------------------------------------

    reset() {
        this.activeModal.close('Reset');
        this.parent.applyFilter(null);
    }
    //---------------------------------------------------------
    apply() {
        this.activeModal.close('Apply');
        // Apply range filters
        //Template_Start_FilterField_Range_3
        this.result.[FilterRangeName] = { from: this.[FilterRangeName]From, to: this.[FilterRangeName]To };
        //Template_End_FilterField_Range_3

        this.parent.applyFilter(this.result);
    }
}
