@Component({

               NgbAlertModule, NgbProgressbarModule, RouterLink, ExportComponent,
               NgxEchartsDirective],
    providers: [provideEchartsCore({ echarts })],
    standalone: true,

})
export class [ComponentNameCapital][ReportTypeCapital]Component         

    extends BaseReportComponent<I[ComponentNameCapital]Filter, I[ComponentNameCapital]WithLookup> {
    override apiService: [SystemNameCapital]Service = inject([SystemNameCapital]Service);
    override tokenService: TokenService = inject(TokenService);
    override queryRequest = {} as I[ComponentNameCapital]Request;
    override exportRequest = {} as I[ComponentNameCapital]Request;
 //   override list: IQueryColumn ; //I[ComponentNameCapital]WithLookup[] = [];
    override downloadFileName: string = "[ComponentMenuName]";

    //------------------------------------------------------
    constructor(modalService: NgbModal, router: Router, excelService: ExcelExportService, activatedRoute: ActivatedRoute) {
        // Initialize queryRequest with default values
        super(modalService, router, excelService, activatedRoute);

        //Template_Start_Component_Report_1
        this.isButton.addNew = false;
        this.getApiUrl = '/[SystemPrefixSmall]/reports/[ComponentNameCapital][ReportTypeCapital]';  
        //Template_End_Component_Report_1

        //Template_Start_Component_Table_1
        this.isButton.chart = false;
        this.addNewRecordLink = { route: "/[SystemPrefixSmall]/[ComponentFileName]/add/0", displayText: "Add New [ComponentMenuName]" };
        this.getApiUrl = '/[ComponentNameCapital]/[ReportTypeCapital]';
        //Template_End_Component_Table_1

        this.filterComponent = [ComponentNameCapital]FilterComponent;
        this.queryRequest = init[ComponentNameCapital]Request();
    }
    //---------------------------------------------------------
    override render(record: IQueryColumn, column: IColumns): any {
        const value = record[column.fieldName as keyof IQueryColumn];
        switch (column.fieldName) {
            //Template_Start_Field_Lookup_1
            case '[ParentTableSmall][CapitalFieldName]':
                return record['[LookupNameSmall]Name?'].toString();
            //Template_End_Field_Lookup_1
            //Template_Start_ReportField_Aggregate_1
            case '[AggregateName]':
                return record['[AggregateName]?'].toString();
            //Template_End_ReportField_Aggregate_1
            default:
                return value;
        }
        return value;
    }
    //---------------------------------------------------------

//Template_Start_Component_Table_2
    override getRecordLinks(record: IQueryColumn): ViewLink[] {
        let actions = get[ComponentNameCapital]Actions(record);
        let links: ViewLink[] = actions.filter(action => 
               action.actionType == 'FrontendLink'
            && action.actionLocation == 'ListRow'
            ).map(action => {
            return { recordId: action.recordId, route: action.route?? '', displayText: action.displayText}
        });

        return links;
    }
    //---------------------------------------------------------
    override getRecordActions(record: IQueryColumn): ActionLink[] {
        let actions = get[ComponentNameCapital]Actions(record);
        let links: ActionLink[] = actions.filter(action => 
               action.actionType == 'FrontendFunction'
            && action.actionLocation == 'ListRow'
            ).map(action => {
            return { recordId: action.recordId, action: action.action?? null, displayText: action.displayText, data: action.data}
        });

        return links;
    }
//Template_End_Component_Table_2
//--------------------------------------------------------------
//Template_Start_Component_Report_2
    override getChart(records: I[ComponentNameCapital]WithLookup[]): EChartsOption {
        // return this.getDemoChart();
        // reorder records in reverse order to show same order of table records.
        let reversedRecords = records.reverse();
        let baseChart = this.getBaseChart();
        baseChart.yAxis = {
        //Template_Start_ChartField_Vertical_1
            data: reversedRecords.map(x => x['[ParentTableSmall][CapitalFieldName]' as keyof I[ComponentNameCapital]WithLookup] ?? "unknown"),
        //Template_End_ChartField_Vertical_1

            type: 'category',
            axisLine: {
                lineStyle: {
                    type: 'dashed', color: getColor('light')
                }
            },
            axisLabel: {
                show: true, color: getColor('body-color')
            },
            splitLine: {
                lineStyle: {
                    color: "rgba(133, 141, 152, 0.1)", type: 'dashed'
                }
            }
        };

        baseChart.series = [     
        //Template_Start_ChartField_Horizontal_1
        {
            name: '[DisplayName]',

            type: 'bar',
            barWidth: "10px",
            barGap: "0.25",
            z: 10,
            itemStyle: {
                borderRadius: [4, 4, 0, 0], color: getColor('info')
            },
            data: reversedRecords.map(x => x.[AggregateName]),
        }
        //Template_End_ChartField_Horizontal_1
        ]
        ;

        return baseChart;
    }
//Template_End_Component_Report_2
}
